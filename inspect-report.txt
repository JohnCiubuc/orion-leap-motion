#!/bin/bash
# preinst script for leapd

set -e

# Create a non-root user with the right privileges.
addgroup leap || true
adduser --system --ingroup leap --no-create-home --home /nonexistent leap || true
usermod -a -G plugdev leap # Access to USB devices

# Permit `leap` to write logs
LOG_DIR=/var/log/ultraleap/
mkdir -p $LOG_DIR
chown -R leap:leap $LOG_DIR

# Permit `leap` to read/write config files
CONFIG_DIR=/etc/ultraleap/
mkdir -p $CONFIG_DIR
chown -R leap:leap $CONFIG_DIR

# Permit `leap` to read ldat files

===== service: postinst =====
#!/bin/bash
# postinst script for leapd
#
# see: dh_installdeb(1)
 
set -e
 
# summary of how this script can be called:
#        * <postinst> `configure' <most-recently-configured-version>
#        * <old-postinst> `abort-upgrade' <new version>
#        * <conflictor's-postinst> `abort-remove' `in-favour' <package>
#          <new-version>
#        * <postinst> `abort-remove'
#        * <deconfigured's-postinst> `abort-deconfigure' `in-favour'
#          <failed-install-package> <version> `removing'
#          <conflicting-package> <version>
# for details, see http://www.debian.org/doc/debian-policy/ or
# the debian-policy package

# Source debconf library.
. /usr/share/debconf/confmodule
 
case "$1" in
    configure)
        # Work around the fact our config may have disappeared despite dpkg
        # marking the package as configured and removed, by providing a
        # suitable configuration file if one does not exist.
        if [ ! -e /etc/ultraleap/hand_tracker_config.json ]; then
          cp /usr/TrackingService/bin/hand_tracker_config.json /etc/ultraleap/hand_tracker_config.json
          chown leap:leap /etc/ultraleap/hand_tracker_config.json
        fi

        # Cleanup temporary install directory
        rm -rf /usr/TrackingService/

        # We made a release where we changed the permissions of this file so everyone could access,
        # we want to change this back, so if the file exists we change it to leap user
        if [ -e /etc/ultraleap/analytics_config.json ]; then
          chown leap:leap /etc/ultraleap/analytics_config.json
          chmod 644 /etc/ultraleap/analytics_config.json
        fi

        if [ -f "$(which udevadm)" ]; then
            # Make udev reload its rules to accept the new rule
            echo -n "Reloading udev rules..."
            udevadm control --reload && echo "done."
        else
            echo "No udevadm detected, skipping reloading udev rules."
        fi

        # Make sure the `leap` user has access to config files that were installed.
        CONFIG_DIR=/etc/ultraleap/
        chown -R leap:leap $CONFIG_DIR

        if [ -f "$(which systemctl)" ]; then
            if systemctl status > /dev/null; then
                systemctl daemon-reload
                systemctl enable ultraleap-hand-tracking-service

                echo -n "Starting leapd as a service... "
                systemctl start ultraleap-hand-tracking-service && echo "done."

                db_get ultraleap-hand-tracking-service/accept_licenses
                if [ "$RET" = "true" ]; then
                    if [ -f "$(which leapctl)" ]; then
                        # wait for the server to be fully setup before trying to use leapctl
                        sleep 5s

                        leapctl eula -y
                    fi
                fi
            else
                echo "Systemctl is not running, skipping leapd service setup."
            fi
        else
            echo "No systemctl detected, skipping leapd service setup."
        fi

    # Make sure everyone can read and write this file specifically
    chmod a+rw $CONFIG_DIR/hand_tracker_config.json

    # Links for SDK standard locations
    mkdir -p /usr/lib/x86_64-linux-gnu/ /usr/include/
    for f in libLeapC.so libLeapC.so.5
    do
      ln -s /usr/lib/ultraleap-hand-tracking-service/$f /usr/lib/x86_64-linux-gnu/$f || true
    done
    ;;
 
    abort-upgrade|abort-remove|abort-deconfigure)
    ;;
 
    *)
        echo "postinst called with unknown argument \`$1'" >&2
        exit 1
    ;;
esac
 
ldconfig
db_stop
 
exit 0

===== service: prerm =====
#!/bin/bash

set -e

if [ -f "$(which systemctl)" ]; then
    if systemctl status > /dev/null; then
        systemctl stop ultraleap-hand-tracking-service || true
        systemctl disable ultraleap-hand-tracking-service || true
    fi
fi

# Remove SDK standard location links
for f in libLeapC.so libLeapC.so.5
do
  rm -f /usr/lib/x86_64-linux-gnu/$f
done

rm -f /usr/include/LeapC.h

# kill all command shouldn't be necessary but there's no harm in calling it
# Give the daemon a second to shut down gracefully
(killall -q -INT leapd && sleep 1) || true
